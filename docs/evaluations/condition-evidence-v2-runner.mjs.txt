function deriveRelation(result) {
  const before=result.source.conditionQuote,after=result.candidate.conditionQuote;
  if ((!before || !after) && result.sameCondition!==null) return undefined;
  if (before && after && typeof result.sameCondition!=='boolean') return undefined;
  if (!before && !after) return 'none';
  if (!before) return 'added';
  if (!after) return 'removed';
  return result.sameCondition ? 'preserved' : 'replaced';
}
// Diagnostic restriction: only a complete leading condition + unchanged action.
// Structural checks are not proof that the model's semantic relation is correct.
function composeEvidence(input, result) {
  for (const [side, text] of [['source', input.sourceText], ['candidate', input.candidateText]]) {
    const {conditionQuote, actionQuote} = result[side];
    if (!text.includes(conditionQuote) || !text.includes(actionQuote)) return {reason:'quote-mismatch'};
    const assembled = conditionQuote ? conditionQuote.trim()+' '+actionQuote.trim() : actionQuote.trim();
    if (text.trim() !== assembled) return {reason:'incomplete-or-overlapping-evidence'};
  }
  if (result.source.actionQuote !== result.candidate.actionQuote) return {reason:'action-also-changed'};
  const before=result.source.conditionQuote, after=result.candidate.conditionQuote;
  const allowed = !before && !after ? ['none'] : !before ? ['added'] : !after ? ['removed'] : ['preserved','replaced'];
  if (!allowed.includes(result.relation) || (before===after && result.relation==='replaced')) return {reason:'relation-evidence-conflict'};
  const action = result.source.actionQuote;
  const render = {
    removed: `원문은 ‘${before}’라는 조건을 붙여 ‘${action}’라고 표현합니다. 후보에서는 그 조건 없이 같은 행동·상태를 표현합니다. 원래 조건을 유지하려면 해당 구절이 있는지 확인해 주세요.`,
    added: `원문은 별도 조건 없이 ‘${action}’라고 표현합니다. 후보에는 ‘${after}’라는 조건이 붙었습니다. 그 조건을 추가하려는 의도인지 확인해 주세요.`,
    replaced: `‘${action}’에 붙는 조건이 ‘${before}’에서 ‘${after}’로 달라졌습니다. 원래 조건을 유지하려면 ‘${before}’의 뜻이 남아 있는지 확인해 주세요.`,
    preserved: `‘${before}’를 ‘${after}’로 다듬었습니다. 조건의 뜻과 ‘${action}’라는 표현은 유지됩니다.`,
    none: `조건 변화는 없으며 ‘${action}’라는 표현이 유지됩니다.`,
  }[result.relation];
  if (render.length>400) return {reason:'length'};
  return {rendered:render,semanticRelationVerified:false};
}
